                                Java Client SDK (`com.evora.client`)
                                
                            

                            
// 1. Instantiate EvoraClient
EvoraClient client = EvoraClient.create("http://localhost:8080");

// 2. Submit job to queue
JobRequest request = JobRequest.builder()
    .queue("critical")
    .priority(1)
    .idempotencyKey("ORDER-9912")
    .payload("{\"action\": \"PROCESS_PAYMENT\", \"amount\": 149.99}")
    .build();

Job job = client.enqueue(request);

// 3. Start background worker node
EvoraWorker worker = EvoraWorker.builder()
    .client(client)
    .queue("critical")
    .concurrency(4)
    .handler(j -> {
        System.out.println("Processing job payload: " + j.getPayload());
        return JobResult.success();
    })
    .build();

worker.start();
                            

                        


                        
                        

                            

                                Spring Boot Integration
                                
                            

                            
@Configuration
public class EvoraConfig {
    @Bean
    public EvoraClient evoraClient(@Value("${evora.url}") String evoraUrl) {
        return EvoraClient.create(evoraUrl);
    }

    @Bean(destroyMethod = "stop")
    public EvoraWorker orderWorker(EvoraClient client, OrderService orderService) {
        EvoraWorker worker = EvoraWorker.builder()
            .client(client)
            .queue("orders")
            .concurrency(4)
            .handler(job -> {
                orderService.process(job.getPayload());
                return JobResult.success();
            })
            .build();
        worker.start();
        return worker;
    }
}
                            

                        


                        
                        

                            

                                cURL REST API
                                
                            

                            
# Submit Job
curl -X POST http://localhost:8080/api/v1/jobs \
  -H "Content-Type: application/json" \
  -d '{
    "queue": "critical",
    "priority": 1,
    "idempotencyKey": "PAY-8812",
    "payload": {"action": "DISPATCH", "user_id": "usr-12"}
  }'

# Poll Job (Worker Node)
curl -X GET "http://localhost:8080/api/v1/jobs/poll?worker_id=worker-01"

# Complete Job
curl -X POST http://localhost:8080/api/v1/jobs/{job_id}/complete \
  -H "Content-Type: application/json" \
  -d '{"worker_id": "worker-01"}'
                            

                        


                        
                        

                            

                                Node.js / TypeScript REST Integration
                                
                            

                            
import axios from 'axios';

const EVORA_URL = 'http://localhost:8080/api/v1';

// Enqueue Job
async function enqueueJob(queue, payload, idempotencyKey) {
  const { data } = await axios.post(`${EVORA_URL}/jobs`, {
    queue,
    priority: 1,
    idempotencyKey,
    payload
  });
  return data;
}

// Poll & Process Worker Loop
async function startWorker(workerId) {
  while (true) {
    try {
      const res = await axios.get(`${EVORA_URL}/jobs/poll?worker_id=${workerId}`);
      const job = res.data;
      console.log('Processing job:', job.id);
      await axios.post(`${EVORA_URL}/jobs/${job.id}/complete`, { worker_id: workerId });
    } catch (err) {
      await new Promise(r => setTimeout(r, 1000));
    }
  }
}
